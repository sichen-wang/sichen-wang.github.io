{{/* 卡片预览摘要： 1) 优先显式标识 card_summary/summary/description/abstract； 2) 否则从正文提取——跳过标题/分隔线/图片/公式块/表格/短代码/裸 HTML， 按“整段”累加，直到再加一段就会超出长度预算为止（尽量多放几段、但不超预算）； 3) 正文没有可用段落时，回退为全文纯文本截断，确保预览非空。 注意：中文需按“字符”而非“字节”计长，故用 strings.RuneCount。 */}} {{ $summary := .Params.card_summary | default .Params.summary | default .Params.description | default .Params.abstract | default "" }} {{ if not $summary }} {{ $budget := 140 }} {{ $raw := .RawContent }} {{/* 统一换行：源文件为 CRLF 时，按 "\n\n" 分段会失效，先归一化 */}} {{ $raw = replaceRE `\r\n?` "\n" $raw }} {{/* 去掉代码块 */}} {{ $raw = replaceRE "(?s)```.*?```" "\n\n" $raw }} {{/* 有 摘要分隔符则只取其前 */}} {{ if strings.Contains $raw "" }} {{ $before := strings.TrimSpace (index (split $raw "") 0) }} {{ if $before }}{{ $raw = $before }}{{ end }} {{ end }} {{ range (split $raw "\n\n") }} {{ $block := strings.TrimSpace . }} {{ $skip := or (not $block) (hasPrefix $block "#") (hasPrefix $block "---") (hasPrefix $block "![") (hasPrefix $block "\\[") (hasPrefix $block "$$") (hasPrefix $block "|") (hasPrefix $block "{{") (hasPrefix $block "<") }} {{ if not $skip }} {{ if not $summary }} {{ $summary = $block }} {{ else if le (add (strings.RuneCount $summary) (strings.RuneCount $block)) $budget }} {{ $summary = printf "%s\n\n%s" $summary $block }} {{ end }} {{ end }} {{ end }} {{/* 兜底：整篇没有可用段落（如全是公式/图片）时，取全文纯文本 */}} {{ if not $summary }} {{ $plain := plainify (replaceRE `(?m)^#+\s*` "" $raw) }} {{ $plain = strings.TrimSpace (replaceRE `\s+` " " $plain) }} {{ if $plain }}{{ $summary = $plain }}{{ end }} {{ end }} {{/* 单段本身过长时截断 */}} {{ if gt (strings.RuneCount $summary) (add $budget 80) }} {{ $summary = printf "%s…" (substr $summary 0 $budget) }} {{ end }} {{ end }} {{ $summary = strings.TrimSpace $summary }} {{ return $summary }}
